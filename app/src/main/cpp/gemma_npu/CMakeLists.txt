cmake_minimum_required(VERSION 3.18)
project(MyTTSGemmaNpu LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(LLAMA_SOURCE "" CACHE PATH "llama.cpp source at the runtime's commit (headers only)")
set(RUNTIME_DIR "" CACHE PATH "Pinned prebuilt llama.cpp / ggml-hexagon arm64 libraries")
foreach(lib llama ggml ggml-base)
  add_library(${lib} SHARED IMPORTED)
  set_target_properties(${lib} PROPERTIES IMPORTED_LOCATION "${RUNTIME_DIR}/lib${lib}.so")
endforeach()
add_library(gemma_npu SHARED gemma_npu.cpp)
target_include_directories(gemma_npu PRIVATE "${LLAMA_SOURCE}/include" "${LLAMA_SOURCE}/ggml/include")
target_link_libraries(gemma_npu PRIVATE llama ggml ggml-base log)
target_link_options(gemma_npu PRIVATE "-Wl,-z,max-page-size=16384")
