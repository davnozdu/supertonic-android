cmake_minimum_required(VERSION 3.22.1)
project(pockettts_android LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../.." ABSOLUTE)
add_compile_options("-ffile-prefix-map=${ROOT}=." "-fdebug-prefix-map=${ROOT}=.")
include(FetchContent)
set(SPM_ENABLE_SHARED OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_TCMALLOC OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_NFKC_COMPILE OFF CACHE BOOL "" FORCE)
FetchContent_Declare(sentencepiece GIT_REPOSITORY https://github.com/google/sentencepiece.git GIT_TAG v0.2.1 GIT_SHALLOW TRUE)
FetchContent_Declare(dr_libs GIT_REPOSITORY https://github.com/mackron/dr_libs.git GIT_TAG 50bb723e6a459dbb781e26cefee4fd9ca6714d6a)
FetchContent_MakeAvailable(sentencepiece dr_libs)
foreach(tool spm_decode spm_encode spm_export_vocab spm_normalize spm_train)
    if(TARGET ${tool})
        target_link_libraries(${tool} log)
    endif()
endforeach()
add_library(pockettts_jni SHARED "${ROOT}/vendor/pockettts/PocketTTS.cpp/pocket_tts.cpp" "${ROOT}/vendor/pockettts/sonic/sonic.c" jni_bridge.cpp)
target_compile_definitions(pockettts_jni PRIVATE PTT_SHARED_LIB)
target_include_directories(pockettts_jni PRIVATE "${ROOT}/build/pocket-headers" "${ROOT}/vendor/pockettts/sonic" "${sentencepiece_SOURCE_DIR}/src" "${dr_libs_SOURCE_DIR}")
target_link_libraries(pockettts_jni PRIVATE "${ROOT}/app/build/onnxruntime/jni/arm64-v8a/libonnxruntime.so" sentencepiece-static log android)
target_link_options(pockettts_jni PRIVATE "-Wl,-z,max-page-size=16384")
